**Refinements to the spec, found while sizing the geometry:**
1. **The pastern slopes up and *back* from the hoof to the fetlock** (anatomy), not forward. Planted angle 0.6 rad; it folds to −0.8 rad in the swing, tucking the hoof back.
2. **Legs step at a cadence.** The view's pace, 57 px of ground per TAU of walk, is too long a stride for these legs. Leg phase = `walk × cadence`, with `cadence = 1.8 × 33 / H`, so smaller beasts step faster. The IK pair is 1.10× the rest distance, which keeps every planted hoof reachable.
3. **Four-beat offsets are `{hf: 0, ff: 0.75, hn: 0.5, fn: 0.25}`.** Lift-off happens at `(DUTY − offset)`, so a higher offset lifts earlier. These offsets give the true lateral order LH, LF, RH, RF.
4. **Leg joints use generic keys** `root`, `top` (elbow / stifle), `mid` (knee / hock), `fetlock`, `hoof` for both leg types.
5. **The car floor is at the car's bottom edge, `y = −16`**, 3 px above the axle, so the existing car shapes sit on it unchanged.
6. **`FkMounts.quadruped` returns the joint Dictionary.** `mounted()` uses `j.saddle` and `j.bob`, so the rider's spine absorbs the real back bob.

```gdscript
class_name FkQuadruped
extends RefCounted
## Joint math for four-legged beasts (horse, deer, boar, ram, bear): a gait-driven spine and head, and legs
## that plant their hooves by two-bone IK. Local space: feet at y = 0, facing +x. Proportions come from a
## beast preset (FkMounts.BEASTS: L = half the body's length, H = leg height).

## Ground covered per radian of walk phase (px) unless the pose gives its own (pose.pace): a planted
## hoof slides back at exactly this pace in the body's frame, so it stays put on the ground.
const PACE := 1.0 / 0.11
## Leg strides per radian of walk phase for a beast of height 33; smaller beasts step faster.
const CADENCE := 1.8
## Share of a stride a hoof spends planted.
const DUTY := 0.65
## Four-beat lateral walk as stride offsets (a higher offset lifts off earlier): left hind, left fore,
## right hind, right fore. Unmirrored, the right legs are the near ones (n), the left the far ones (f).
const LEGS := {"hf": 0.0, "ff": 0.75, "hn": 0.5, "fn": 0.25}
## Rest position of the poll (head joint) per head kind, relative to the barrel centre plus L along x,
## and how far the head thrusts on the charge (× lunge).
const HEAD := {"horse": [Vector2(12, -25), 0.4], "deer": [Vector2(10, -29), 0.4], "boar": [Vector2(4, -2), 0.5],
	"ram": [Vector2(5, -8), 0.6], "bear": [Vector2(6, -8), 0.6]}
## The IK pair's length over the rest distance from the top joint to the fetlock: soft legs, and every
## planted hoof within reach.
const SLACK := 1.10


static func cadence(bp: Dictionary) -> float:
	return CADENCE * 33.0 / float(bp.H)


static func solve(bp: Dictionary, pose: Dictionary, seed := 0) -> Dictionary:
	var L: float = bp.L
	var H: float = bp.H
	var mv := FkPaint.move_amount(pose)
	var t: float = pose.get("t", 0.0)
	var cad := cadence(bp)
	var theta: float = pose.get("walk", 0.0) * cad
	var pace: float = pose.get("pace", PACE)
	var lunge := maxf(0.0, FkUnits.swing(pose.get("atk", -1.0))) * 5.0
	var bob := lerpf(sin(t * 1.5 + seed) * 0.5, absf(sin(theta)) * 2.0, mv)
	var c := Vector2(lunge, -H - 10 + bob)
	# The back pitches with the gait: withers leading, croup lagging.
	var pitch := sin(theta * 2.0 + 0.5) * mv
	var head_kind: String = bp.head
	var j := {"c": c, "lunge": lunge, "bob": bob, "cadence": cad,
		"withers": c + Vector2(0.55 * L, -11 + pitch), "croup": c + Vector2(-0.6 * L, -9 - pitch)}
	j["back"] = (j.withers as Vector2).lerp(j.croup, 0.5)
	j["dock"] = (j.croup as Vector2) + Vector2(-5, -2)
	j["saddle"] = c + Vector2(-2, -10 - (5.0 if head_kind == "bear" else 0.0) - pitch * 0.3)
	# The head nods twice a stride, as a walking horse's does.
	var hk: Array = HEAD.get(head_kind, HEAD.horse)
	var nod := sin(theta * 2.0) * mv
	j["head_angle"] = nod * 0.06
	j["poll"] = c + (hk[0] as Vector2) + Vector2(L + lunge * float(hk[1]), nod * 1.2)
	# Legs: the top joint rides the body; the hoof follows the gait; knee (fore) or hock (hind) by IK.
	var span := pace / cad * DUTY * TAU
	var P := 0.14 * H
	var w := 0.3 * L
	var mirrored: bool = pose.get("mirrored", false)
	var z := {}
	var legs := {}
	for leg in LEGS:
		var front: bool = leg[0] == "f"
		var near: bool = leg[1] == "n"
		var root := c + Vector2(0.62 * L * (1.0 if front else -1.0), 5)
		var top_off := Vector2(0, 0.22 * H) if front else Vector2(0.12 * H, 0.28 * H)
		var top := root + top_off
		# At rest the fetlock stands straight under the top joint, the pastern sloping up and back.
		var rest_pastern := Vector2(-sin(0.6), -cos(0.6)) * P
		var hoof_x0 := top.x - rest_pastern.x
		var rest_top_y := -H - 5 + top_off.y
		var pair := SLACK * absf(rest_pastern.y - rest_top_y)
		var l1 := pair * (0.52 if front else 0.5)
		var l2 := pair - l1
		var s := fposmod(theta / TAU + LEGS[leg], 1.0)
		var planted := s < DUTY
		var gx: float
		var lift := 0.0
		var fold := 0.0
		if planted:
			gx = span * (0.5 - s / DUTY)
		else:
			var u := (s - DUTY) / (1.0 - DUTY)
			gx = span * (u - 0.5)
			lift = sin(u * PI) * (0.18 if front else 0.14) * H
			fold = sin(u * PI)
		var hoof := Vector2(hoof_x0 + gx * mv, -lift * mv)
		var a := lerpf(0.6, 0.6 - 1.4 * fold, mv)
		var pastern := Vector2(-sin(a), -cos(a)) * P
		var fet := FkSkeleton.reach(top, hoof + pastern, l1, l2)
		hoof = fet - pastern
		legs[leg] = {"root": root, "top": top, "mid": FkSkeleton.ik(top, fet, l1, l2, FkSkeleton.KNEE if front else FkSkeleton.ELBOW),
			"fetlock": fet, "hoof": hoof, "planted": planted or mv == 0.0}
		z[leg] = (w if near else -w) * (-1.0 if mirrored else 1.0)
	j["legs"] = legs
	j["z"] = z
	return j
```

## Chariot layers, back to front: the driver stands inside the car, its near panel hiding them below the
## thigh, the near wheel over the panel.
const CHARIOT_ORDER := ["pole", "far wheel", "car back", "driver", "car front", "near wheel", "yoke"]
## Ground covered per radian of walk phase for a chariot (px), and the wheel's radius.
const CHARIOT_PACE := 1.0 / 0.12
const WHEEL_R := 13.0
## Where the driver stands (x) in the car.
const DRIVER_X := -17.0


## Wheel joints (center = axle, top, bottom = ground contact), the wheel's spin (ground covered over the
## radius: no slipping), the car floor's height and where the driver stands (bumped by the ride).
static func chariot_joints(pose: Dictionary) -> Dictionary:
	var center := Vector2(-18, -WHEEL_R)
	var angle: float = pose.get("walk", 0.0) * CHARIOT_PACE / WHEEL_R
	var floor := -16.0
	var bump := sin(angle * 6.0) * 0.6 * FkPaint.move_amount(pose)
	return {"center": center, "top": center + Vector2(0, -WHEEL_R), "bottom": center + Vector2(0, WHEEL_R),
		"angle": angle, "floor": floor, "driver": Vector2(DRIVER_X, floor + bump)}


## The car's side silhouette per kind (its bottom edge is the floor).
static func _car_outline(kind: String) -> Array:
	match kind:
		"wood":
			return [Vector2(-30, -16), Vector2(-6, -16), Vector2(0, -30), Vector2(-4, -40), Vector2(-12, -30), Vector2(-28, -28)]
		"iron":
			return [Vector2(-30, -14), Vector2(-4, -14), Vector2(-4, -32), Vector2(-30, -32)]
		_:
			return [Vector2(-30, -16), Vector2(-6, -16), Vector2(-4, -36), Vector2(-26, -34)]


static func chariot(ci: CanvasItem, st: Dictionary, pose: Dictionary, seed: int) -> void:
